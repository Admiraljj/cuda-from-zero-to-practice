cmake_minimum_required(VERSION 3.24)
project(cuda_from_zero_to_practice LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(BUILD_CUDA_EXAMPLES "Build the published CUDA examples" ON)

function(cpu_example target source)
  add_executable(${target} ${source} ${ARGN})
  if(BUILD_TESTING)
    add_test(NAME ${target} COMMAND ${target})
  endif()
endfunction()
